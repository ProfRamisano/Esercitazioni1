\begin{esercizio}
Data l'equazione: 
\[
P(z)=z^2+\lambda z-3 (1+2i)=0
\]
Determinare il valore di $\lambda\in \mathbb{C}$ tale che $z_1=2-i$ sia radice dell'equazione. \\
Calcolare inoltre, per tale valore di $\lambda$, anche l'altra radice dell'equazione data. 
\end{esercizio}
\textit{Soluzione.} Essendo radice, $z_1$ soddisfa l'equazione: 
\[
(2-i)^2+\lambda \, (2-i)-3(1+2i)=0
\quad \Rightarrow \quad 
\lambda=\frac{10i}{2-i}=2 \, (-1+2i)
\] 
La seconda radice si ottiene osservando che è nota la somma delle soluzioni dell'equazione: 
\[
z_1+z_2=-\lambda
\quad \Rightarrow \quad 
z_2=-\lambda-z_1=-2 \, (-1+2i)-(2-i)=-3i
\]

\begin{esercizio}
	Individuare sul piano di Gauss il luogo $\Gamma$ dei punti immagine dei complessi per i quali la seguente funzione assume valori reali:
	\[
	w(z)=i \, |z|^2+2z
	\]
\end{esercizio}
\textit{Soluzione.} Posto $z=x+iy$:
\[
w(z)=i \, (x^2+y^2)+2(x+iy)=2z+i \, (x^2+y^2+2y)
\]
Da cui: 
\[
\Gamma: \, x^2+y^2+2y=0
\]
Si tratta di una circonferenza di centro $(0,-1)$ e raggio $R=1$. 

\begin{esercizio}
	Determinare i complessi $z$ immaginari puri che verificano l'uguaglianza:
	\[
	|z-i|=|z+2|
	\]
\end{esercizio}
\textit{Soluzione.} \' E possibile interpretare geometricamente l'uguaglianza in termini di distanze. 
\begin{itemize}
	\item $|z+2|$ rappresenta la distanza dei punti $P$ da $A(-2,0)$
	\item $|z-i|$ rappresenta la distanza dei punti $P$ da $B(0,1)$
\end{itemize}
L'uguaglianza è il luogo dei punti $P(x,y)$ equidistanti da $(0,1)$ e da $(-2,0)$. 
\begin{figure}[h!!]
	\centering
	\includegraphics[scale=0.3]{assecomplessi1.PNG}
	\end{figure}
La soluzione del problema è il punto $P$ di intersezione tra l'asse del segmento $AB$ e l'asse delle ordinate. Le intersezioni possono essere dedotte con metodi di geometria analitica oppure algebricamente.\\
\newline
Sia $z=iy$ immaginario puro:
\begin{align*}
|z-i|=|z+2| \quad \Rightarrow \quad 
|iy-i|=|2+iy| \quad \Rightarrow \quad 
|i(y-1)|=|2+iy|
\end{align*}
Si ha: 
\[
(y-1)^2=4+y^2 \quad \Rightarrow \quad 
y=-\frac{3}{2}
\]

\begin{esercizio}
	Individuare la regione del piano di Gauss luogo delle immagini dei complessi che soddisfano la disuguaglianza: 
	\[
	|z-i|\leq |z+2|
	\]
\end{esercizio}
\textit{Soluzione}. Geometricamente, l'uguaglianza descrive l'asse del segmento di  estremi $A(-2,0)$ e $B(0,1)$.\\
La disuguaglianza $|z-i|<|z+2|$ richiede di considerare anche i punti nel semipiano individuati dall'asse che sono più vicini a $(0,1)$. 

\begin{esercizio}
	Individuare la regione del piano di Gauss luogo delle immagini dei complessi che soddisfano la disuguaglianza: 
	\[
	|\sqrt{2}z+i|\leq |z-1|
	\]
\end{esercizio}
\textit{Soluzione}. Posto $z=x+iy$:
\begin{align*}
	|\sqrt{2}z+i|\leq |z-1| \quad \Rightarrow \quad 
	&|\sqrt{2} \, x+i(\sqrt{2}y+1)|\leq |(x-1)+iy|\\
	&x^2+y^2+2\sqrt{2} y+2x\leq 0 
	\quad \Rightarrow \quad 
	\Gamma: \, (x+1)^2+(y+\sqrt{2})^2\leq 3
\end{align*}
Il luogo descritto è il disco chiuso di centro $C(-1, -\sqrt{2})$ e raggio $R=\sqrt{3}$. 

\begin{esercizio}
	Determinare i numeri complessi che soddisfano la relazione: 
	\[
	|z-1|+i|z|=|z-i|+i
	\]
\end{esercizio}
\textit{Soluzione.} I complessi sono uguali se e sono se sono uguali tra loro rispettivamente le parti reali e quelle immaginarie: 
\[
\begin{cases}
	|z-1|=|z-i|\\
	|z|=1
\end{cases}
\]
Il sistema può essere risolto geometricamente intersecando l'asse del segmento di estremi $A(1,0)$ e $B(0,1)$ con la circonferenza di centro l'origine e raggio unitario. Trattando le equazioni dal punto di vista della geometria analitica:
\[
\begin{cases}
	y=x\\
	x^2+y^2=1
\end{cases}
\Rightarrow \quad 
\Big (\pm \frac{\sqrt{2}}{2}, \pm \frac{\sqrt{2}}{2}\Big )
\]

\begin{esercizio}
	Determinare il luogo dei punti immagini delle soluzioni dell'equazione: 
	\[
	|\overline{z}-i| \cdot |z|=|\overline{z}-i|^2
	\]
\end{esercizio}
\textit{Soluzione}. Si osserva che $\overline{z}=i$ è soluzione. \\
Posto $z=-i$ una soluzione dell'equazione, è possibile semplificare il fattore comune: 
\[
|z|=|\overline{z}-i|
\quad \Rightarrow \quad 
|z|=|\overline{\overline{z}-i}|
\quad \Rightarrow \quad 
|z|=|z+i|
\]
L'uguaglianza descrive l'asse del segmento di estremi $A(0,0)$ e $B(0,-1)$: si tratta della retta $y=-1/2$.

\begin{esercizio}
	Calcolare modulo e argomenti dei numeri complessi: 
	\[
	z_1=(1-i)(-1+\sqrt{3} \, i)
	\qquad 
	z_2=\frac{-1+\sqrt{3} \, i}{1-i}
	\qquad 
	(1-i)^{1976}
	\]
\end{esercizio} 
\textit{Soluzione}. 
\begin{itemize}
	\item $z_1$ è prodotto tra due numeri complessi che possono essere espressi in forma trigonometrica: 
	\[
	z_1=\Big (\sqrt{2}, -\frac{\pi}{4}\Big )\cdot \Big (2, \frac{2}{3} \, \pi)=\Big (2\sqrt{2}, -\frac{\pi}{4}+\frac{2}{3} \, \pi\Big )
	=\Big (2\sqrt{2}, \frac{5}{12}\, \pi\Big )
	\]
	\item Analogamente per quanto riguarda $z_2$:
	\[
	z_2=\dfrac{\Big (2, \dfrac{2}{3} \, \pi \Big )}{\Big (\sqrt{2}, -\dfrac{\pi}{4}\Big )}=\Big (\sqrt{2}, \frac{2}{3} \, \pi +\frac{\pi}{4}\Big )=\Big (\sqrt{2}, \frac{11}{12} \, \pi \Big )
	\]
	\item La base $(1-i)$ viene espressa in forma trigonometria: 
	\[
	(1-i)=\Big (\sqrt{2}, -\frac{\pi}{4}\Big )
	\]
	Si ha: 
	\[
	(1-i)^{1976}=\Big (\sqrt{2}, -\frac{\pi}{4}\Big )^{1976}=\Big (\big (\sqrt{2}\big )^{1976}, 1976\cdot \Big (-\frac{\pi}{4}\Big ) \Big )=
	\big (2^{988}, -494\pi \big ) \equiv \big (2^{988}, 0\big ) 
	\]
\end{itemize}

\begin{esercizio}
	Determinare modulo e argomento principale ($-\pi\leq \theta\leq \pi$) del numero complesso: 
	\[
	z=\frac{1}{1-i \, \tan \phi}
	\qquad 
		-\pi\leq \phi<\pi \quad , \quad 
		\phi\neq \pm \pi/2
	\]
\end{esercizio}
\textit{Soluzione.} Posto $z=\rho \, (\cos \theta+i\sin \theta)$:
\begin{align*}
	z&=\frac{1}{1-i \, \tan \phi}=\dfrac{1}{1-i \, \dfrac{\sin \phi}{\cos \phi}}=\frac{\cos \phi}{\cos \phi-i\sin \phi}
	=\cos \phi \, (\cos \phi +i\ \sin \phi)
	\end{align*}
	Per esprimere $z$ in forma trigonometrica occorre rendere positivo il coefficiente contenete $\cos \phi$:
	\begin{itemize} 
		\item Se $\cos \phi>0$, cioè se $-\pi/2<\phi<\pi/2$:
		\[
		z=\cos \phi \, (\cos \phi +i\ \sin \phi)
		\]
		\item Se $\cos \phi<0$, cioè se $-\pi<\phi<-\pi/2$ o se $\pi/2<\phi<\pi$:
		\[
		z=-\cos \phi \, (-\cos \phi -i\ \sin \phi)=-\cos \phi \, \big (\cos (\pi+\phi)+i \, \sin (\pi+\phi)\big )
		\]
		\end{itemize} 
Si osserva che $\pi+\phi$ non è un angolo principale e, quindi, occorre sfruttare la periodicità per esprimerlo nell'intervallo ammissibile: 
\[
\pi+\phi\equiv (\pi+\phi)-2\pi=\phi-\pi 
\]
Confrontando: 
\begin{align*}
	\rho &=\begin{cases} 
		\cos \phi & \textrm{se $-\pi/2<\phi<\pi/2$}\\
		-\cos \phi &\textrm{se $-\pi<\phi<-\pi/2$ o se $\pi/2<\phi<\pi$}
\end{cases}
\\
\theta&=
\begin{cases} 
	\phi & \textrm{se $-\pi/2<\phi<\pi/2$}\\
	\phi-\pi &\textrm{se $-\pi<\phi<-\pi/2$ o se $\pi/2<\phi<\pi$} 
	\end{cases} 
\end{align*}

\begin{esercizio}
	Risolvere l'equazione: 
	\[
	z^3=\overline{z}^2
	\]
\end{esercizio}
\textit{Soluzione.} Sia $z=\rho \, (\cos \theta+i\sin \theta)$:
\begin{align*}
\rho^3 \, (\cos 3\theta+i\sin 3\theta)&=\rho^2 \, \big (\cos (-2\theta)+i\sin (-2\theta)\big )\\
\end{align*}
Uguagliando i termini omologhi: 
\[
\begin{cases}
	\rho^3=\rho^2\\
3\theta=-2\theta+k2\pi 	
\end{cases}
\Rightarrow \quad 
\begin{cases}
	\rho=1\\
	\theta=k \, \dfrac{2\pi }{5} 
\end{cases}
\]
Esistono sei soluzioni distinte, ottenute rispettivamente per $k=0,1,2,3,4,5$. 

\begin{esercizio}
	Risolvere nei naturali l'equazione: 
	\[
	(-\sqrt{3}+i)^n=4\, (1-i\sqrt{3})
	\]
\end{esercizio}
\textit{Soluzione.} Si esprimono i due membri in forma trigonometrica: 
\[
\Big (2, \frac{5\pi}{6}\Big )^n=4 \, \Big (2, -\frac{\pi}{3}\Big )
\quad \Rightarrow \quad 
\Big (2^n, \frac{5n\pi}{6}\Big )=\Big (8, -\frac{\pi}{3}\Big )
\]
Si ha: 
\[
\begin{cases}
	2^n=8\\[6pt]
\dfrac{5n\pi}{6}=-\dfrac{\pi}{3}+k2\pi
\end{cases}
\Rightarrow \quad 
\begin{cases}
	n=3\\[6pt]
	\dfrac{5}{2} \, \pi+\dfrac{\pi}{3}=k2\pi 
\end{cases}
\]
Non esistono $k\in \mathbb{Z}$ che soddisfano la seconda equazione: 
\[
\forall \, k \in \mathbb{Z} \qquad 
\frac{17}{6} \, \pi \neq k2\pi 
\]

\begin{esercizio}
	Risolvere nei naturali l'equazione: 
	\[
	(-\sqrt{3}+i)^n=-8\, (1+i\sqrt{3})
	\]
\end{esercizio}
\textit{Soluzione.} Si esprimono i due membri in forma trigonometrica: 
\[
\Big (2, \frac{5\pi}{6}\Big )^n=8 \, \Big (2, \frac{4\pi}{3}\Big )
\quad \Rightarrow \quad 
\Big (2^n, \frac{5n\pi}{6}\Big )=\Big (16, \frac{4\pi}{3}\Big )
\]
Si ha: 
\[
\begin{cases}
	2^n=16\\[6pt]
	\dfrac{5n\pi}{6}=\dfrac{4\pi}{3}+k2\pi
\end{cases}
\Rightarrow \quad 
\begin{cases}
	n=4\\[6pt]
	\dfrac{10}{3} \, \pi-\dfrac{4\pi}{3}=k2\pi 
\end{cases}
\]
Si risolve la seconda equazione ricercando una soluzione $k\in \mathbb{Z}$:
\[
\dfrac{10}{3} \, \pi-\dfrac{4\pi}{3}=k2\pi 
\quad \Rightarrow \quad 
2\pi=k2\pi \quad \Rightarrow \quad 
k=1 \in \mathbb{Z}
\]
Si conclude che l'equazione ammette soluzione $n=4$. 

\begin{esercizio}
	Dato $\omega=-1+i\sqrt{3}$, determinare i complessi $z$ tali che $\omega z$ sia reale negativo. 
\end{esercizio}
\textit{Soluzione.} Si esprime $\omega$ in forma trigonometrica: 
\[
\omega=2 \, \Big (-\frac{1}{2}+\frac{\sqrt{3}}{2} \, i\Big )=\Big (2, \frac{2\pi}{3}\Big )
\]
Posto $z=(\rho, \theta)$:
\[
z\omega=\Big (2\rho, \frac{2\pi}{3}+\theta\Big)
\]
Si pongono le condizioni:
\[
\begin{cases}
	\dfrac{2\pi}{3}+\theta=\pi+k2\pi \\[6pt]
	\rho>0
\end{cases}
\Rightarrow \quad 
\begin{cases}
	\theta=\dfrac{\pi}{3}+k2\pi\\[6pt]
	\rho>0
\end{cases}
\Rightarrow \quad 
z=\rho \, \Big (\frac{1}{2}+\frac{\sqrt{3}}{2} \, i\Big )
\]
Geometricamente, le soluzioni descrivono la semiretta aperta nel primo quadrante, passante per l'origine e con inclinazione $\pi/3$ rispetto al semiasse positivo. 

\begin{esercizio}
	Si consideri il numero complesso: 
	\[
	\alpha=-\sqrt{3}-i
	\]
	Determinare i complessi $z$ tali che $\beta=z/\alpha$ abbiano immagini sulla retta di equazione $y=x/\sqrt{3}$.
\end{esercizio}
\textit{Soluzione.} Siano $z$ e $\alpha$ espressi in forma trigonometrica: 
\[
z=(\rho, \theta) \qquad \alpha=\Big (2, \frac{7\pi}{6}\Big )
\]
Si ha: 
\[
\beta=\frac{z}{\alpha}=\Big (\frac{\rho}{2}, \theta-\frac{7\pi}{6}\Big )
\]
Il numero $\beta$ ha immagine sulla retta se e solo se $\tan \beta=1/\sqrt{3}$, ossia se $\beta=\pi/6+k\pi$:
\[
\theta-\frac{7\pi}{6}=\frac{\pi}{6}+k\pi 
\quad \Rightarrow \quad 
\theta=\frac{4\pi }{3} +k\pi\equiv \frac{\pi}{3}+k\pi   
\]
Da cui: 
\[
z=\Big (\rho , \frac{\pi}{3}+k\pi\Big )
\qquad \rho\geq 0
\]
Il luogo degli $z$ è la retta passante per l'origine e inclinata di $\frac{\pi}{3}$. 

\begin{esercizio}
	Applicare le formule di De Moivre per ricavare le formule di $\sin(n\theta)$ e di $\cos(n\theta)$: 
\end{esercizio}
\textit{Soluzione.} Applicando le formule di De Moivre: 
\[
(\cos \theta+i\sin \theta)^n=\cos(n\theta)+i\sin (n\theta)
\]
Applicando il binomio di Newton: 
\[
(\cos \theta+i\sin \theta)^n =\sum_{k=0}^n {n\choose k} \cos^{n-k} \theta\cdot i^{k} \, \sin^{k}\theta
\]
Si sfrutta la proprietà delle potenze di $i$ per uguagliare le componenti omologhe delle due espressioni: 
\begin{align*}
\cos(n\theta)&=\cos^n \theta-{n\choose 2}\cos^{n-2}\theta\cdot \sin^2\theta+
{n\choose 4}\cos^{n-4}\theta\cdot \sin^4 \theta-\dots=\sum_{k=0}^{\lfloor n/2\rfloor} (-1)^k {n\choose 2k} \cos(\theta)^{n-2k}\sin(\theta)^{2k}\\
\sin(n\theta)&={n\choose 1} \cos^{n-1}\theta\sin \theta-
{n\choose 3} \cos^{n-3}\theta\sin^3 \theta+\dots=\sum_{k=0}^{\lfloor (n-1)/2\rfloor } (-1)^k {n\choose 2k+1} \cos(\theta)^{n-(2k+1)}\sin(\theta)^{2k+1}
\end{align*}

\begin{esercizio}
	Calcolare le radici cubiche del numero complesso: 
	\[
	z=\frac{1}{(1+i)^2}
	\]
\end{esercizio}
\textit{Soluzione.} Si esprime il numero complesso in forma trigonometrica: 
\[
(1+i)=\Big (\sqrt{2}, \frac{\pi}{4}\Big )
\quad \Rightarrow \quad 
(1+i)^{-2}=\Big (\frac{1}{2}, -\frac{\pi}{2}\Big )
\]
Si ha: 
\[
z_k=\sqrt[3]{\frac{1}{2}} \, \Big (\cos \Big (\frac{-\pi/2+k2\pi}{3}\Big )+i\sin \Big (\frac{-\pi/2+k2\pi}{3}\Big )\Big )
\]
Da cui le tre radici cubiche complesse: 
\begin{align*}
	z_0&=\sqrt[3]{\frac{1}{2}} \, \Big (\cos \Big (-\frac{\pi}{6}\Big )+i\sin \Big (-\frac{\pi}{6}\Big )\Big )=\frac{\sqrt{3}-i}{2\sqrt[3]{2}}\\
	z_1&=\sqrt[3]{\frac{1}{2}} \, \Big (\cos \Big (-\frac{\pi}{6}+\frac{2\pi}{3}\Big )+i\sin \Big (-\frac{\pi}{6}+\frac{2\pi}{3}\Big )\Big )=\frac{i}{\sqrt[3]{2}}\\
	z_2&=\sqrt[3]{\frac{1}{2}} \, \Big (\cos \Big (-\frac{\pi}{6}+\frac{4\pi}{3}\Big )+i\sin \Big (-\frac{\pi}{6}+\frac{4\pi}{3}\Big )\Big )=
	-\frac{\sqrt{3}+i}{2\sqrt[3]{2}}
\end{align*}
Geometricamente, le intersezioni sono ai vertici di un triangolo equilatero avente un vertice $(0, 1/\sqrt[3]{2})$. \\

\begin{esercizio}
	Scrivere in forma algebrica una delle radici quarte di $(1-2i)^2$. 
\end{esercizio}
\textit{Soluzione.} Si nota, in generale, che esistono esattamente $n$ radici $n$-esime di un numero complesso e che, per questo motivo, non è consentita alcuna eventuale operazione di semplificazione tra l'indice della radice e la l'esponente del radicando. \\
Tuttavia in questo esercizio non si richiede di individuare tutte le radici e si osserva che una radice quarta di $(1-2i)^2$ coincide con una radice quadrata di $1-2i$:
\[
1-2i=\rho \, \big (\cos \theta+i\sin \theta) \quad 
\Rightarrow \quad 
\sqrt{1-2i}=\pm \sqrt{\rho} \, \left (\cos \Big (\frac{\theta+k2\pi}{2}\Big )+i\sin \Big (\frac{\theta+k2\pi}{2}\Big )\right ) 
\qquad k=0,1
\] 
Osservando che $-\pi/2<\theta<0$, ossia $-\pi/4<\theta/2<0$:
\[
 \begin{cases}
 	\rho=\sqrt{5}\\
	\cos \theta=\dfrac{1}{\sqrt{5}}\\[9pt]
	\sin \theta=\dfrac{-2}{\sqrt{5}}
\end{cases}
\Rightarrow \quad 
\begin{cases}
	\rho=\sqrt{5}\\[6pt]
	\cos \dfrac{\theta}{2}=\dfrac{1+1/\sqrt{5}}{2}=\dfrac{1}{\sqrt[4]{5}} \, \sqrt{\dfrac{\sqrt{5}+1}{2}}\\[9pt]
	\sin \dfrac{\theta}{2}=-\sqrt{\dfrac{1-1/\sqrt{5}}{2}}=-\dfrac{1}{\sqrt[4]{5}} \, \sqrt{\dfrac{\sqrt{5}-1}{2}}
\end{cases}
\]
Posto $k=0$:
\[
z_0=\sqrt{\frac{\sqrt{5}+1}{2}}-i \, \sqrt{\frac{\sqrt{5}-1}{2}}
\]
Le immagini delle restanti tre radici si trovano sulla circonferenza di raggio $\sqrt[4]{5}$ e sui vertici del quadrato avente un vertice in $z_0$. 

\begin{esercizio}
	Una delle radici quarte di un numero complesso $z$ risulta: 
	\[
	z_0=\cos 2 -i\sin 2
	\]
	Determinare le altre radici quarte di $z$. 
\end{esercizio}
\textit{Soluzione.} Le radici quarte del numero complesso sono rappresentate sul piano di Gauss ai vertici di un quadrato inscritto in una circonferenza di raggio uguale al modulo di una delle radici. \\
In questo caso la circonferenza ha raggio unitario e per il calcolo delle restanti radici è sufficiente incrementare di volta in volta l'angolo di $\pi/2$. \\
Occorre rappresentare $z_0$ in forma goniometrica: 
\[
z_0=\cos2-i\sin 2=1 \, \big (\cos(-2)+i\sin(-2)\big )
\]
Si ha:
\[
z_0=(1, -2)
\quad \Rightarrow \quad 
\begin{cases}
z_1=\Big (1, -2+\dfrac{\pi}{2}\Big )=\sin 2+i\cos 2\\[6pt]
z_2=(1, -2+\pi)=-\cos 2+i\sin 2\\[6pt]
z_3=\Big (1, -2+\dfrac{3\pi}{2}\Big )=-\sin 2-i\cos 2
\end{cases}
\]

\begin{esercizio}
	Trovare le soluzioni della seguente equazione: 
	\[
	(-1+i)^3 z^3=1
	\]
\end{esercizio}
\textit{Soluzione.} Risolvendo rispetto a $z$:
\[
z=\sqrt[3]{\frac{1}{(-1+i)^3}}
\]
I valori di $z$ si ottengono associando in tutti i modi possibili alle tre radici del numeratore le tre radici del denominatore. Dei nove valori trovati, solo tre sono tra loro distinti. Segue che le tre radici distinte si possono ottenere associando alle tre radici distinte del numeratore una sola, qualsiasi, radice del denominatore:
\[
z=\frac{\sqrt[3]{1}}{-1+i}
\qquad \quad \sqrt[3]{1}=\Big (1, \frac{2k\pi}{3}\Big )
=\begin{cases}
	z_0=1\\
	z_1=-\dfrac{1}{2}+\dfrac{\sqrt{3}}{2} \, i\\[6pt] 
	z_2=-\dfrac{1}{2}-\dfrac{\sqrt{3}}{2} \, i
\end{cases}
%\qquad -1+i=\Big (\sqrt{2}, \frac{3\pi}{4}\Big )
\]
Si ha:
\begin{align*}
	z_0&=\frac{1}{-1+i}=\frac{1}{2} \, (-1-i)\\
	z_1&=\dfrac{-1/2+i\sqrt{3}/2}{-1+i}=\dfrac{1+\sqrt{3}}{4}+i \, \dfrac{1-\sqrt{3}}{4}\\
	z_2&=\dfrac{-1/2-i\sqrt{3}/2}{-1+i}=\dfrac{1-\sqrt{3}}{4}+i \, \dfrac{1+\sqrt{3}}{4}
\end{align*}

Si osserva che le soluzioni ottenute sono a due a due coniugate, coerentemente con il fatto che l'equazione complessa è a coefficienti reali. 

\begin{esercizio}
	Risolvere l'equazione:
	\[
	p(z)=z^4+iz^3-(1-i)^6z-i(1-i)^6=0
	\]
\end{esercizio}
\textit{Soluzione.} Si osserva che $z=-i$ è soluzione. Fattorizzando con il metodo di Ruffini si ottiene:
\[
p(z)=(z+i)\big (z^3-(1-i)^6\big )=0
\]
Si deduce che le radici dell'equazione sono $-i$ e le tre radici cubiche del numero $(1-i)^6$.
Occorre quindi calcolare le tre radici cubiche. Si osserva che una risulta $(1-i)^2=-2i$. Le altre sono poste ai vertici del triangolo equilatero avente un vertice corrispondente a $-2i$ e inscritto in una circonferenza di raggio $2$:
\[
z_k=2 \, \left (\cos \Big (\frac{-\pi/2+k2\pi}{3}\Big )+i\sin \Big (\frac{-\pi/2+k2\pi}{3}\Big )\right )
\]
Da cui, oltre alle soluzioni $z_4=-i$ e $z_3=-2i$, si ha: 
\begin{align*}
&z_1=2 \, \Big (\cos \frac{\pi}{6}+i\sin \frac{\pi}{6}\Big )=\sqrt{3}+i
&z_2=2 \, \Big (\cos \frac{5\pi}{6}+i\sin \frac{5\pi}{6}\Big )=-\sqrt{3}+i
\end{align*}

\begin{esercizio}
	Risolvere in campo complesso la seguente equazione: 
	\[
	\sqrt[3]{z}=\sqrt{\sqrt{3}+i}
	\]
\end{esercizio}
\textit{Soluzione.} L'equazione è in campo complesso: è possibile elevare entrambi i membri alla sesta potenza: 
\[
z^2=(\sqrt{3}+i)^3=8i
\quad \Rightarrow \quad 
z=\sqrt{8i}=2\sqrt{2} \cdot  \sqrt{i}=\pm \sqrt{8} \, \Big (\cos \frac{\pi}{4}+i\sin \frac{\pi}{4}\Big )=\pm 2 \, (1+i)
\]
Si osserva che l'operazione di elevamento è consentita in $\mathbb{C}$:
\[
\sqrt[3]{z}=\sqrt{\alpha} \quad \Leftrightarrow \quad 
z=\big (\sqrt{\alpha}\big )^3=\sqrt{\alpha^3} \quad \Leftrightarrow \quad 
z^2=\alpha^3
\]
In $\mathbb{R}$ invece l'elevamento non avrebbe portato a scritture equivalenti: 
\[
\sqrt[3]{x}=\sqrt{\alpha} \quad \Leftrightarrow \quad 
x=\sqrt{\alpha^3} \quad \Rightarrow \quad 
x^2=\alpha^3
\]
Infatti l'implicazione contraria non è soddisfatta: se $x^2=\alpha^3$, per ogni $x$ e con $\alpha\geq 0$, sarebbe falso $x=\sqrt{\alpha^3}$, per la non-negatività della radice quadrata. 

\begin{esercizio}
	Sia data la seguente equazione in $\mathbb{C}$:
	\[
	z^8=\frac{2|z|^2+\lambda}{2+|z|} \, |z|^7
	\]
	Determinare per quali valori di $\lambda\in \mathbb{R}^+$ esistono soluzioni. 
\end{esercizio}
\textit{Soluzione.} $z=0$ è soluzione per ogni $\lambda$. Si osserva inoltre che il secondo membro è un numero reale positivo per ogni $z$: la potenza ottava di $z$ è un numero reale positivo.\\
Se $z$ è soluzione allora $|z|$ soddisfa l'uguaglianza: 
\[
|z|^8=\frac{2|z|^2+\lambda}{2+|z|} \, |z|^7
\quad \Rightarrow \quad 
|z|=0 \, \, \, \vee \, \, \, |z|=\frac{2|z|^2+\lambda}{2+|z|} 
\quad \Rightarrow \quad 
|z|^2-2|z|+\lambda=0
\]
Si ha: 
\[
|z|=1\pm \sqrt{1-\lambda}
\]
\begin{itemize}
	\item Se $\lambda>1$ allora l'unica soluzione è quella banale $z=0$
	\item Se $\lambda=1$ allora $|z|=1$. Sostituendo nell'equazione data:
	\[
	z^8=\frac{2+1}{2+1}=1
	\]
	Le soluzioni sono le $8$ radici dell'unità:
	\[
	z^8=1 
	\quad \Rightarrow \quad 
	z_k=\cos\frac{k\pi}{4}+i\sin \frac{k\pi}{4} \qquad 0\leq k<8
	\]
	\item Se $0<\lambda<1$: 
	\[\rho_1:=|z_1|=1+\sqrt{1-\lambda} \qquad \quad 
	\rho_2:=|z_2|=1-\sqrt{1-\lambda}
	\]
	Le soluzioni sono del tipo $z_1=\rho_1 u_1$ e $z_2=\rho_2 u_2$, dove $u_1, u_2$ sono complessi unimodulari.  \\
	In quanto soluzioni, soddisfano l'equazione data:
	\[
	(\rho_1 u_1)^8=\frac{2\rho_1^2+\lambda}{2+\rho_1} \, \rho_1^7
	\quad \Rightarrow \quad 
	\rho_1^8 u_1^8=\frac{2\rho_1^2+\lambda}{2+\rho_1} \, \rho_1^7
	\quad \Rightarrow \quad 
	u_1^8=\frac{2\rho_1^2+\lambda}{\rho_1 \, (2+\rho_1)}
	\]
	Dall'uguaglianza precedente: 
	\[
	\rho_1^2-2\rho_1+\lambda=0 
	\quad \Rightarrow \quad 
	\begin{cases}
		\rho_1^2=2\rho_1-\lambda\\[6pt]
		\rho_1=\dfrac{\rho_1^2+\lambda}{2}
	\end{cases}
	\Rightarrow \quad 
	\begin{cases}
		\lambda=2\rho_1-\rho_1^2\\[6pt]
		\rho_1=\dfrac{\rho_1^2+\lambda}{2}
	\end{cases}
	\]
	Si ha: 
	\[
	u_1^8=\frac{2\rho_1^2+\lambda}{2\rho_1+\rho_1^2}=
\frac{2\rho_1^2+(2\rho_1-\rho_1^2)}{2\rho_1+\rho_1^2}=1 \quad \Rightarrow \quad 
u_1^8=1	
	\]
	Da cui si deducono otto soluzioni: 
	\[
	z=\rho_1 \sqrt[8]{1} 
	\quad \Rightarrow \quad 
	z_k=\rho_1 \, \Big (\cos \frac{k\pi}{4}+\sin \frac{k\pi}{4}\Big )
	\qquad \quad 0\leq k<8
	\]
	In modo analogo nel caso di $\rho_2 u_2$:
	\[
	z=\rho_2 \sqrt[8]{1} 
	\quad \Rightarrow \quad 
	z_k=\rho_2 \, \Big (\cos \frac{k\pi}{4}+\sin \frac{k\pi}{4}\Big )
	\qquad \quad 0\leq k<8
	\]
\end{itemize}
